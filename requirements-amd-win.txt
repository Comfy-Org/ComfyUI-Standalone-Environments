# AMD GPU requirements (ROCm 7.14, Windows)
# Versions are pinned to prevent pip from preferring newer CPU-only wheels from PyPI.
#
# ROCm 7.14 ships via AMD's TheRock multi-arch wheel index, which serves both
# Windows and Linux from the same URL - the old repo.radeon.com universal
# package (last: rocm-rel-7.2.1, Python 3.12 only) is no longer used. Wheels
# cover Python 3.10-3.14, so this build now uses the workspace default
# Python instead of a pinned 3.12. The [device-all] extra pulls in the
# matching ROCm device libraries for all supported GPU architectures.

--index-url https://repo.amd.com/rocm/whl-multi-arch/
--extra-index-url https://pypi.org/simple

torch[device-all]==2.12.0+rocm7.14.0
torchvision[device-all]==0.27.0+rocm7.14.0
torchaudio==2.11.0+rocm7.14.0
